function SignalBadge({ signal }: { signal: string }) {
  const upper = signal.toUpperCase();
  if (upper.includes('BUY')) {
    return <span className="px-2 py-0.5 rounded text-xs font-bold bg-green-600 text-white">{signal}</span>;
  }
  if (upper.includes('SELL')) {
    return <span className="px-2 py-0.5 rounded text-xs font-bold bg-red-600 text-white">{signal}</span>;
  }
  return <span className="px-2 py-0.5 rounded text-xs font-bold bg-gray-500 text-white">{signal}</span>;
}
