    ```typescript
    // --- Chart types (Phase 13) ---

    export type ChartTimeframe = '1D' | '1M' | '3M' | '6M' | '1Y';

    export interface AlpacaBar {
      t: string;  // ISO timestamp
      o: number;  // open
      h: number;  // high
      l: number;  // low
      c: number;  // close
      v: number;  // volume
      n: number;  // number of trades
      vw: number; // volume-weighted average price
    }

    export interface ChartOverlay {
      ticker: string;
      analysis_date: string;
      signal: string;
      entry_price: number | null;
      take_profit: number | null;
      stop_loss: number | null;
      expiry_date: string | null;
      strategy_name: string | null;
      options_legs: string;
      final_trade_decision: string;
    }

    export const TIMEFRAME_CONFIG: Record<ChartTimeframe, { alpacaTimeframe: string; daysBack: number; label: string }> = {
      '1D': { alpacaTimeframe: '15Min', daysBack: 0, label: '1D' },
      '1M': { alpacaTimeframe: '1Day', daysBack: 30, label: '1M' },
      '3M': { alpacaTimeframe: '1Day', daysBack: 90, label: '3M' },
      '6M': { alpacaTimeframe: '1Day', daysBack: 180, label: '6M' },
      '1Y': { alpacaTimeframe: '1Day', daysBack: 365, label: '1Y' },
    };
    ```

    **3. Create frontend/src/hooks/useChartData.ts:**

    Hook signature: `useChartData(ticker: string, timeframe: ChartTimeframe)`
    Returns: `{ bars: CandlestickData[] | null, volumeData: HistogramData[] | null, loading: boolean, error: string | null }`

    Implementation details:
    - Use `useRef(new Map<string, { candles: CandlestickData[], volumes: HistogramData[] }>())` for client-side cache keyed by `${ticker}-${timeframe}` (per D-21)
    - On cache hit: set data immediately, skip fetch
    - On cache miss: fetch from `https://data.alpaca.markets/v2/stocks/${ticker}/bars` with headers `APCA-API-KEY-ID: import.meta.env.VITE_ALPACA_KEY` and `APCA-API-SECRET-KEY: import.meta.env.VITE_ALPACA_SECRET`
    - Query params: `timeframe=${alpacaTimeframe}&start=${startDate}&feed=iex&limit=1000&sort=asc`
    - For `1D` preset: start = today's date in YYYY-MM-DD format. For others: start = today minus daysBack
    - Transform AlpacaBar[] to CandlestickData[]: `{ time: bar.t.slice(0, 10) as Time, open: bar.o, high: bar.h, low: bar.l, close: bar.c }` for daily bars. For 15Min (1D): use Unix timestamp `Math.floor(new Date(bar.t).getTime() / 1000)` as time
    - Transform to HistogramData[]: `{ time: (same as candle), value: bar.v, color: bar.c >= bar.o ? '#22c55e33' : '#ef444433' }` (semi-transparent green/red)
    - Store in cache after successful fetch
    - useEffect dependency array: `[ticker, timeframe]`
    - Handle fetch errors: set error string, loading false
    - Handle empty ticker: skip fetch, return null data
