```python
def _parse_tabular_string(s: str) -> Optional[pd.DataFrame]:
    if not s:
        return None
    stripped = s.strip()
    if not stripped or stripped.lower().startswith("no "):
        return None
    try:
        return pd.read_csv(io.StringIO(stripped), sep=r'\s+', engine='python')
    except Exception:
        return None
```
